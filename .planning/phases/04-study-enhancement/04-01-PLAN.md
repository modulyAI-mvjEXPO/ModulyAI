```typescript
const MARK_INSTRUCTIONS: Readonly<Record<string, string>> = {

  '5M': 'The student is preparing for a 5-mark question. Give a clear explanation with 1 illustrative example. Use a short list or diagram description if helpful. Aim for 150-200 words.',

  '10M': 'The student is preparing for a 10-mark question. Give a comprehensive, well-structured answer with multiple examples, comparisons where relevant, and a conclusion. Use markdown headings and lists. Aim for 400-500 words.',
};

const markInstruction = mark && MARK_INSTRUCTIONS[mark]
  ? `\n${MARK_INSTRUCTIONS[mark]}`
  : mark
    ? `\nThe student is preparing for a ${mark} question. Calibrate answer depth appropriately.`
    : '';
```

Place `MARK_INSTRUCTIONS` as a module-level constant, just before the `buildSystemPrompt` function.
