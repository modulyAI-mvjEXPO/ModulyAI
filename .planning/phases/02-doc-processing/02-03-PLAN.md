function formatSize(bytes: number | null): string {
  if (bytes === null || bytes === undefined) return '';
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;

**Rewritten `UploadDocs` component:**

```typescript
export function UploadDocs() {
    const [documents, setDocuments] = useState<ReadonlyArray<DocumentRow>>([]);
    const [loading, setLoading] = useState(true);
    const [fetchError, setFetchError] = useState('');
    const [userId, setUserId] = useState<string | null>(null);
    const pollingRef = useRef<ReturnType<typeof setInterval> | null>(null);

    // Get current user
    useEffect(() => {
        const getUser = async () => {
            const { data: { user } } = await supabase.auth.getUser();
            if (user) setUserId(user.id);
        };
        getUser();
    }, []);

    // Fetch documents from Supabase
    const fetchDocuments = useCallback(async () => {
        if (!userId) return;
        try {
            const { data, error } = await supabase
                .from('documents')
                .select('*')
                .eq('user_id', userId)
                .order('created_at', { ascending: false });

            if (error) throw error;
            setDocuments(data as ReadonlyArray<DocumentRow>);
            setFetchError('');
        } catch (e) {
            console.error('Error fetching documents:', e);
            setFetchError('Could not load documents.');
        } finally {
            setLoading(false);
        }
    }, [userId]);

    // Initial fetch when userId is available
    useEffect(() => {
        if (userId) fetchDocuments();
    }, [userId, fetchDocuments]);

    // Poll for processing documents
    useEffect(() => {
        const hasProcessing = documents.some(doc => doc.status === 'processing');

        if (hasProcessing && !pollingRef.current) {
            pollingRef.current = setInterval(() => {
                fetchDocuments();
            }, 3000);
        } else if (!hasProcessing && pollingRef.current) {
            clearInterval(pollingRef.current);
            pollingRef.current = null;
        }

        return () => {
            if (pollingRef.current) {
                clearInterval(pollingRef.current);
                pollingRef.current = null;
            }
        };
    }, [documents, fetchDocuments]);

    // Handle new upload — refetch to include the new document
    const handleUploadSuccess = (_newFile: UploadedFile) => {
        // Refetch from Supabase to get the actual document row
        fetchDocuments();
    };

    // Format date for display
    const formatDate = (isoDate: string): string => {
        const d = new Date(isoDate);
        return `Uploaded on ${d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })}`;
    };
```
